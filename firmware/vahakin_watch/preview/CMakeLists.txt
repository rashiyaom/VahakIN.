cmake_minimum_required(VERSION 3.16)
project(vahakin_watch_preview C CXX)
set(CMAKE_CXX_STANDARD 17)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../lv_conf.h" CACHE STRING "" FORCE)
set(LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
file(GLOB_RECURSE LV_SOURCES "${LVGL_DIR}/src/*.c")
add_library(lvgl STATIC ${LV_SOURCES})
target_include_directories(lvgl PUBLIC "${LVGL_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_definitions(lvgl PUBLIC LV_CONF_INCLUDE_SIMPLE)
add_executable(render render.cpp)
target_link_libraries(render PRIVATE lvgl)
